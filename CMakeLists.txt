cmake_minimum_required(VERSION 3.16)
project(ORBYX VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ORBYX_BUILD_GUI "Build Qt 6 desktop interface" ON)
option(ORBYX_INSTALL_BATCH "Install the optional Python 3.10+ batch runner" ON)
option(ORBYX_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ORBYX_FUZZ "Build the libFuzzer parser harness (Clang only)" OFF)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs video)
find_package(jsoncpp CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(orbyx_core src/core/engine.cpp)
target_include_directories(orbyx_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(orbyx_core PUBLIC ${OpenCV_LIBS} JsonCpp::JsonCpp Threads::Threads)
target_compile_options(orbyx_core PRIVATE -Wall -Wextra -Wpedantic)
if(ORBYX_SANITIZE)
  target_compile_options(orbyx_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(orbyx_core PUBLIC -fsanitize=address,undefined)
endif()
if(ORBYX_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "ORBYX_FUZZ requires Clang (libFuzzer); set -DCMAKE_CXX_COMPILER=clang++")
  endif()
  # Instrument the core for coverage-guided fuzzing without pulling in the
  # libFuzzer main (fuzzer-no-link), while still catching memory/UB bugs.
  target_compile_options(orbyx_core PUBLIC -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  target_link_options(orbyx_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(orbyx src/cli/main.cpp)
target_compile_definitions(orbyx PRIVATE ORBYX_VERSION="${PROJECT_VERSION}")
target_link_libraries(orbyx PRIVATE orbyx_core)
if(ORBYX_BUILD_GUI)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
  add_executable(orbyx-gui src/gui/main.cpp)
  target_link_libraries(orbyx-gui PRIVATE orbyx_core Qt6::Widgets)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME batch-regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_batch.py)
  add_test(NAME detect-regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_detect.py)
  add_test(NAME real-data-download-regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fetch_real_rdas.py)
  add_test(NAME sign-regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sign.py)
  add_executable(orbyx_tests tests/test_engine.cpp)
  target_link_libraries(orbyx_tests PRIVATE orbyx_core)
  add_test(NAME engine-regression COMMAND orbyx_tests)
  add_test(NAME color-cli-flow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_color_cli.py $<TARGET_FILE:orbyx_tests> $<TARGET_FILE:orbyx>)
  add_test(NAME sbom-regression COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sbom.py $<TARGET_FILE:orbyx> $<TARGET_FILE:orbyx_tests>)
  add_test(NAME cli-help COMMAND orbyx --help)
  add_test(NAME cli-version COMMAND orbyx --version)
  add_test(NAME cli-invalid COMMAND orbyx --unknown)
  set_tests_properties(cli-invalid PROPERTIES WILL_FAIL TRUE)
  # Synthetic end-to-end simulation: an independently encoded dataset driven
  # through calibrate+generate, recovering the known injected offsets. Exercises
  # the real CLI and the PNG decoder on third-party-encoded input. It does not
  # substitute for real-acquisition / SatDump verification (M3-01).
  add_test(NAME simulation-flow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/orbyx_simulate.py --orbyx $<TARGET_FILE:orbyx>)
  set_tests_properties(simulation-flow PROPERTIES TIMEOUT 300)
  add_test(NAME scaled-calibration-flow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/orbyx_simulate.py --orbyx $<TARGET_FILE:orbyx> --width 253 --shift 5 --feature-margin 24 --calibration-scale 0.5)
  set_tests_properties(scaled-calibration-flow PROPERTIES TIMEOUT 300)
  if(ORBYX_FUZZ)
    add_executable(orbyx_fuzz_parsers tests/fuzz_parsers.cpp)
    target_link_libraries(orbyx_fuzz_parsers PRIVATE orbyx_core)
    target_compile_options(orbyx_fuzz_parsers PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(orbyx_fuzz_parsers PRIVATE -fsanitize=fuzzer,address,undefined)
    # Bounded, deterministic smoke run: builds the harness and proves it survives
    # several thousand mutations without a sanitizer finding. Extended campaigns
    # run separately in CI with -max_total_time.
    add_test(NAME fuzz-parsers-smoke COMMAND orbyx_fuzz_parsers -runs=8000 -max_len=8192 -timeout=10)
    set_tests_properties(fuzz-parsers-smoke PROPERTIES TIMEOUT 180)
  endif()
  if(ORBYX_BUILD_GUI)
    add_test(NAME gui-color-flow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_gui_color.py $<TARGET_FILE:orbyx_tests> $<TARGET_FILE:orbyx-gui> $<TARGET_FILE:orbyx>)
    set_tests_properties(gui-color-flow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
    add_test(NAME gui-product-flow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_gui_flow.py $<TARGET_FILE:orbyx_tests> $<TARGET_FILE:orbyx-gui> $<TARGET_FILE:orbyx>)
    set_tests_properties(gui-product-flow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
    add_test(NAME gui-smoke COMMAND orbyx-gui --smoke-test)
    set_tests_properties(gui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
    add_test(NAME gui-queue-flow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_gui_queue.py $<TARGET_FILE:orbyx_tests> $<TARGET_FILE:orbyx-gui> $<TARGET_FILE:orbyx> ${CMAKE_CURRENT_SOURCE_DIR}/scripts/orbyx_batch.py)
    set_tests_properties(gui-queue-flow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
  endif()
endif()
include(GNUInstallDirs)
if(ORBYX_INSTALL_BATCH)
  install(PROGRAMS scripts/orbyx_batch.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME orbyx-batch)
endif()
install(TARGETS orbyx RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(ORBYX_BUILD_GUI)
  install(TARGETS orbyx-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES resources/orbyx.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES assets/orbyx.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
install(FILES resources/calibration.json DESTINATION ${CMAKE_INSTALL_DATADIR}/orbyx)
install(FILES LICENSE NOTICE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/orbyx)
set(CPACK_PACKAGE_NAME "orbyx")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "https://github.com/chiaraberti13/ORBYX/issues")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "RDAS satellite channel alignment studio")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/chiaraberti13/ORBYX")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
# Declare the licence explicitly so packages do not report "unknown".
set(CPACK_RPM_PACKAGE_LICENSE "GPLv3")
set(CPACK_RPM_PACKAGE_GROUP "Applications/Multimedia")
set(CPACK_RPM_PACKAGE_URL "https://github.com/chiaraberti13/ORBYX")
set(CPACK_DEBIAN_PACKAGE_SECTION "science")
set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/chiaraberti13/ORBYX")
# Native libraries remain distro-resolved through shlibdeps/rpm automatic requirements.
# Python is used by the optional installed script and is invisible to ELF scanners.
if(ORBYX_INSTALL_BATCH)
  if(CPACK_DEBIAN_PACKAGE_DEPENDS)
    string(APPEND CPACK_DEBIAN_PACKAGE_DEPENDS ", python3 (>= 3.10)")
  else()
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.10)")
  endif()
  if(CPACK_RPM_PACKAGE_REQUIRES)
    string(APPEND CPACK_RPM_PACKAGE_REQUIRES ", python3 >= 3.10")
  else()
    set(CPACK_RPM_PACKAGE_REQUIRES "python3 >= 3.10")
  endif()
endif()
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
set(CPACK_GENERATOR "TGZ")
set(CPACK_SOURCE_GENERATOR "TGZ")
set(CPACK_SOURCE_IGNORE_FILES "/build[^/]*/;/[.]git/;~$;[.]user$")
include(CPack)
